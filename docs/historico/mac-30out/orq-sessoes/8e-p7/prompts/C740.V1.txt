+interface Linha {
+  id: string;

+// São Paulo não tem horário de verão: dia local = UTC - 3h.

+export async function avisarPromessas(
+  db: Db,
+  agoraISO: string,
+): Promise<{ promessas: number; avisos_gravados: number; sem_destinatario: number }> {
+  const desde = new Date(Date.parse(agoraISO) - JANELA_VENCIDA_HORAS * MS_HORA).toISOString();
+  const colunas = "id, cliente_id, texto, prazo";
+  const [abertas, vencidas] = await Promise.all([
+    db.from("perf_promessas").select(colunas).eq("status", "aberta"),
+    db.from("perf_promessas").select(colunas).eq("status", "vencida").gte("fechada_em", desde),
+  ]);
+  const hoje = diaSp(agoraISO);
+  const alvos: { p: Linha; sufixo: "hoje" | "vencida" }[] = [
+    ...((abertas.data ?? []) as Linha[]).filter((p) => diaSp(p.prazo) === hoje).map((p) => ({ p, sufixo: "hoje" as const })),
+    ...((vencidas.data ?? []) as Linha[]).map((p) => ({ p, sufixo: "vencida" as const })),
+  ];
+  if (alvos.length === 0) return { promessas: 0, avisos_gravados: 0, sem_destinatario: 0 };
+
+  const quemCuida = await carregarQuemCuida(db);
+  let gravados = 0;
+  let semDestinatario = 0;
+  for (const { p, sufixo } of alvos) {
+    const pessoas = responsaveisDoCliente(quemCuida, p.cliente_id).pessoas;
+    if (pessoas.length === 0) {
+      semDestinatario += 1;
+      continue;
+    }
+    const { data: cli } = await db.from("clientes").select("nome").eq("id", p.cliente_id).maybeSingle();
+    const nome = (cli as { nome?: string | null } | null)?.nome ?? "Cliente";
+    for (const pessoa of pessoas) {
+      const r = await gravarAviso(db, {
+        employeeId: pessoa.id,
+        categoria: "promessa",
+        titulo: nome,
+        mensagem: sufixo === "hoje" ? `Promessa vence hoje: "${p.texto}"` : `Promessa venceu sem entrega: "${p.texto}"`,
+        link: `/dashboard/clientes/${p.cliente_id}`,
+        chave: `promessa:${p.id}:${sufixo}`,
+      });
+      if (r.gravado) gravados += 1;
+    }
+  }
+  return { promessas: alvos.length, avisos_gravados: gravados, sem_destinatario: semDestinatario };
+}

+async function fechar(db: Db, p: Promessa, status: "cumprida" | "vencida", agoraISO: string, extra: Record<string, unknown>): Promise<boolean> {

+    origem: "promessa",
+    chave: `promessa:${p.id}:${status === "cumprida" ? "cumprida" : "vencida"}`,
+    ocorridoEm: agoraISO,

+export async function fecharPromessas(db: Db, agoraISO: string, teto: number = 40): Promise<ResumoFechamento> {

+  const { data } = await db
+    .from("perf_promessas")

+    .eq("status", "aberta");

+  resumo.abertas = promessas.length;
+  if (promessas.length === 0) return resumo;
+
+  const tarefaIds = promessas.map((p) => p.tarefa_id).filter((x): x is string => !!x);
+  const concluidas = new Set<string>();
+  if (tarefaIds.length > 0) {
+    const { data: ts } = await db.from("tarefas_kanban").select("id").in("id", tarefaIds).eq("status", "concluido");
+    for (const t of (ts ?? []) as { id: string }[]) concluidas.add(t.id);
+  }
+
+  const { data: origem } = await db.from("cs_wpp_mensagem").select("wpp_message_id, enviada_em").in("wpp_message_id", promessas.map((p) => p.wpp_message_id));

+  const desde = Array.from(emDaPromessa.values()).sort()[0];
+  let linhas: LinhaMensagem[] = [];
+  if (desde) {
+    const { data: msgs } = await db
+      .from("cs_wpp_mensagem")
+      .select("wpp_message_id, remote_jid, enviada_em, texto, " + COLUNAS_AUTOR)
+      .in("remote_jid", Array.from(new Set(promessas.map((p) => p.group_jid))))
+      .gte("enviada_em", desde)
+      .lte("enviada_em", agoraISO)
+      .order("enviada_em", { ascending: true })
+      .limit(LIMITE_MENSAGENS);
+    linhas = (msgs ?? []) as unknown as LinhaMensagem[];

+        ? linhas
+            .filter((l) => l.remote_jid === p.group_jid && l.enviada_em > em && l.wpp_message_id !== p.wpp_message_id && (l.texto ?? "").trim() && ehDaEquipe(l, telefonesEquipe, mapa))
+            .slice(0, MAX_ENTREGA)
+        : [];

+          funcao: "promessas",
+          system: SYSTEM,
+          prompt: `Promessa: ${p.texto}\n\nMensagens seguintes da equipe:\n` + equipe.map((m) => `[${m.wpp_message_id}] ${(m.texto ?? "").trim()}`).join("\n"),
+          contexto: { cliente_id: p.cliente_id },
+        });
+        const ps = SaidaSchema.safeParse(JSON.parse(r.texto.replace(/^```(?:json)?\s*|\s*```$/g, "")));
+        const alvo = ps.success ? equipe.find((m) => m.wpp_message_id === ps.data.wpp_message_id) : undefined;

+          continue;
+        }
+      }
+      if (Date.parse(p.prazo) < Date.parse(agoraISO) && (await fechar(db, p, "vencida", agoraISO, {}))) resumo.vencidas++;
+    } catch {
+      resumo.falhas++;
+    }
+  }
+  return resumo;
