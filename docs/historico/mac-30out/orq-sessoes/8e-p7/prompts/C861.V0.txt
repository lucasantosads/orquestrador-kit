+    if (pedido && equipe.length) saida.push({ group_jid: grupo, cliente_id: clienteId, pedido, equipe });
+    equipe = [];
+  };
+  for (const m of msgs) {
+    if (m.classe === "desconhecido") continue;
+    if (m.classe === "cliente") {
+      fechar();
+      pedido = m;
+      continue;
+    }
+    if (!pedido || equipe.length >= MAX_EQUIPE || jaTem.has(m.id)) continue;
+    if (minutosUteisEntre(pedido.em, m.em, feriados) > MINUTOS_DIA_UTIL) continue;
+    equipe.push(m);
+  }
+  fechar();
+  return saida;

+export async function detectarPromessas(db: Db, agoraISO: string, teto: number = TETO_CHAMADAS): Promise<ResumoPromessas> {
+  const resumo: ResumoPromessas = { candidatos: 0, chamadas: 0, gravadas: 0, descartadas: 0, falhas: 0 };
+  const { data: gruposData } = await db.from("wpp_grupo").select("group_jid, cliente_id").is("deleted_at", null).neq("status", "inativo");
+  const clientePorGrupo = new Map<string, string>();
+  for (const g of (gruposData ?? []) as { group_jid: string | null; cliente_id: string | null }[]) {
+    if (g.group_jid && g.cliente_id) clientePorGrupo.set(g.group_jid, g.cliente_id);
+  }
+  const jids = Array.from(clientePorGrupo.keys());
+  if (jids.length === 0) return resumo;
+
+  const desde = new Date(Date.parse(agoraISO) - JANELA_HORAS * MS_HORA).toISOString();
+  const { data: msgData } = await db
+    .from("cs_wpp_mensagem")
+    .select("wpp_message_id, remote_jid, enviada_em, texto, " + COLUNAS_AUTOR)
+    .in("remote_jid", jids)
+    .gte("enviada_em", desde)
+    .lte("enviada_em", agoraISO)
+    .order("enviada_em", { ascending: true })
+    .limit(LIMITE_MENSAGENS);
+  const linhas = (msgData ?? []) as unknown as LinhaMensagem[];
+
+  const colaboradores = await carregarTelefonesPorColaborador(db);
+  const telefonesEquipe = colaboradores.map((c) => c.telefone);
+  const feriados = await carregarFeriados(db);
+  const mapa = montarMapaLid(linhas);
+  const contatos = new Map<string, string[]>();
+  for (const id of Array.from(new Set(Array.from(clientePorGrupo.values())))) {
+    contatos.set(id, await contatosEfetivosDoCliente(db as unknown as SupabaseClient, id));
+  }
+
+  const { data: existentes } = await db.from("perf_promessas").select("wpp_message_id").in("wpp_message_id", linhas.map((l) => l.wpp_message_id));
+  const jaTem = new Set<string>();
+  for (const e of (existentes ?? []) as { wpp_message_id: string }[]) jaTem.add(e.wpp_message_id);
+
+  const porGrupo = new Map<string, Msg[]>();
+  const vistas = new Set<string>();
+  for (const l of linhas) {
+    const clienteId = clientePorGrupo.get(l.remote_jid);
+    const texto = (l.texto ?? "").trim();
+    if (!clienteId || !texto || vistas.has(l.wpp_message_id)) continue;
+    vistas.add(l.wpp_message_id);
+    const classe = classificarAutor(l, { telefonesEquipe, contatosCliente: contatos.get(clienteId) ?? [] }, mapa);
+    const lista = porGrupo.get(l.remote_jid) ?? [];
+    lista.push({ id: l.wpp_message_id, em: l.enviada_em, texto, classe, linha: l });
+    porGrupo.set(l.remote_jid, lista);
+  }
+
+  const candidatos: Candidato[] = [];
+  for (const jid of jids) {
+    candidatos.push(...montarCandidatos(jid, clientePorGrupo.get(jid) as string, porGrupo.get(jid) ?? [], jaTem, feriados));
+  }
+  resumo.candidatos = candidatos.length;
+
+  for (const c of candidatos) {
+    if (resumo.chamadas >= teto) break;
+    resumo.chamadas++;
+    try {
+      const r = await chamarProvider({
+        funcao: "promessas",
+        system: SYSTEM,
+        prompt: `Pedido do cliente: ${c.pedido.texto}\n\nRespostas da equipe:\n` + c.equipe.map((m) => `[${m.id}] ${m.texto}`).join("\n"),
+        contexto: { cliente_id: c.cliente_id },
+      });
+      const ps = SaidaSchema.safeParse(JSON.parse(r.texto.replace(/^```(?:json)?\s*|\s*```$/g, "")));
+      const alvo = ps.success ? c.equipe.find((m) => m.id === ps.data.wpp_message_id) : undefined;
+      if (!ps.success || !ps.data.eh_promessa || !alvo || alvo.classe !== "equipe" || !norm(alvo.texto).includes(norm(ps.data.evidencia))) {
+        resumo.descartadas++;
+        continue;
+      }
+      const tel = alvo.linha.chave?.fromMe === true ? undefined : pessoaDaLinha(alvo.linha, mapa)?.telefone;
+      const autor = tel ? colaboradores.find((x) => phonesMatch(tel, x.telefone)) : undefined;
+      const { error } = await db.from("perf_promessas").upsert(
+        {
+          cliente_id: c.cliente_id,
+          group_jid: c.group_jid,
+          wpp_message_id: alvo.id,
+          autor_id: autor?.employee_id || null,
+          texto: ps.data.texto_promessa,
+          prazo: resolverPrazo(ps.data.prazo_iso, alvo.em, feriados),
+          status: "aberta",
+          pedido_wpp_message_id: c.pedido.id,
+          pedido_trecho: c.pedido.texto.slice(0, TRECHO_MAX),
+          evidencia: ps.data.evidencia,
+        },
+        { onConflict: "wpp_message_id", ignoreDuplicates: true },
+      );
+      if (error) resumo.falhas++;
+      else resumo.gravadas++;
+    } catch {
+      resumo.falhas++;
+    }
+  }
+  return resumo;
