+/**
+ * Backfill dos contatos do GHL de um cliente (a rota e o job reusam este laço). Paginado e
+ * re-entrante. Falha nunca vira vazio: todo erro sobe como BackfillErro (status HTTP e
+ * mensagem); falha da busca leva processados, falhas e o cursor para retomar.
+ * Credencial por location (provider='ghl' + location_id): a de outra location nunca serve.
+ */
+import { createClient } from "@supabase/supabase-js";
+import { getEncryptionKey } from "@/lib/perf/encryption-key";
+import { renovarTokenGhl } from "@/lib/perf/ghl-oauth";
+import { ingerirLeadGhl, mapGhlAppContactToLead } from "@/lib/perf/leads-ghl";
+
+export const PREFIXO_ERRO_BACKFILL = "Backfill";
+
+const CONTATOS_POR_PAGINA = 100;
+const MARGEM_RENOVACAO_MS = 5 * 60 * 1000;
+const GHL_API_VERSION = "2021-07-28";
+
+const supabase = createClient(
+  process.env.NEXT_PUBLIC_SUPABASE_URL || "https://placeholder.supabase.co",
+  process.env.SUPABASE_SERVICE_ROLE_KEY || "placeholder",
+);
+
+export type CursorBackfill = (string | number)[];
+
+export type ParcialBackfill = { processados: number; falhas: number; cursor: CursorBackfill | null };
+
+export class BackfillErro extends Error {
+  status: number;
+  mensagem: string;
+  parcial?: ParcialBackfill;
+  constructor(status: number, mensagem: string, parcial?: ParcialBackfill) {
+    super(mensagem);
+    this.name = "BackfillErro";
+    this.status = status;
+    this.mensagem = mensagem;
+    this.parcial = parcial;
+  }
+}
+
+export function lerCursor(v: unknown): CursorBackfill | null {
+  if (!Array.isArray(v) || v.length === 0) return null;
+  return v.every((x) => typeof x === "string" || typeof x === "number") ? (v as CursorBackfill) : null;
+}
+
+function dataOriginal(v: unknown): string | null {
+  if (typeof v !== "string" || !v.trim()) return null;
+  const d = new Date(v);
+  return Number.isNaN(d.getTime()) ? null : d.toISOString();
+}
+
+async function decifrar(cifrado: string, chave: string): Promise<string | null> {
+  const { data, error } = await supabase.rpc("decrypt_meta_token", { encrypted_token: cifrado, encryption_key: chave });
+  return error || !data ? null : (data as string);
+}
+
+export async function credencialDaConta(
+  locationId: string,
+  fetchImpl: typeof fetch = (...a) => fetch(...a),
+): Promise<string> {
+  const { data: cred, error: credErr } = await supabase
+    .from("integration_tokens")
+    .select("access_token, refresh_token, token_expires_at")
+    .eq("provider", "ghl")
+    .eq("location_id", locationId)
+    .limit(1)
+    .maybeSingle();
+  if (credErr) {
+    console.error("[ghl-backfill] leitura da credencial falhou:", credErr.message);
+    throw new BackfillErro(500, "Falha ao ler a credencial do CRM");
+  }
+  if (!cred?.access_token) {
+    throw new BackfillErro(409, "A conta do CRM deste cliente ainda não foi autorizada. Instale o app para importar os contatos.");
+  }
+
+  const chave = getEncryptionKey();
+  if (!chave) {
+    console.error("[ghl-backfill] cifra nao configurada");
+    throw new BackfillErro(500, "Cifra não configurada");
+  }
+
+  let bearer: string | null;
+  const expiraEm = cred.token_expires_at ? new Date(cred.token_expires_at as string).getTime() : null;
+  try {
+    if (expiraEm !== null && expiraEm - Date.now() < MARGEM_RENOVACAO_MS) {
+      const refreshPlano = cred.refresh_token ? await decifrar(cred.refresh_token as string, chave) : null;
+      if (!refreshPlano) throw new Error("credencial de renovação ilegível");
+      bearer = (await renovarTokenGhl(locationId, refreshPlano, fetchImpl)).accessToken;
+    } else {
+      bearer = await decifrar(cred.access_token as string, chave);
+    }
+  } catch (err) {
+    const status = (err as { status?: number })?.status;
+    console.error("[ghl-backfill] renovacao da credencial falhou", status ?? "sem status");
+    throw new BackfillErro(502, "Falha ao renovar a autorização do CRM");
+  }
+  if (!bearer) {
+    console.error("[ghl-backfill] decifrar credencial falhou");
+    throw new BackfillErro(500, "Falha ao ler a credencial do CRM");
+  }
+  return bearer;
+}
+
+export async function backfillContatos(args: {
+  clienteId: string;
+  cursor: CursorBackfill | null;
+  maxPaginas: number;
+  fetchImpl?: typeof fetch;
+}): Promise<{ processados: number; falhas: number; cursor: CursorBackfill | null; terminou: boolean }> {
+  const { clienteId, maxPaginas, fetchImpl = (...a: Parameters<typeof fetch>) => fetch(...a) } = args;
+
+  const { data: loc, error: locErr } = await supabase
+    .from("perf_ghl_locations")
+    .select("location_id")
+    .eq("cliente_id", clienteId)
+    .eq("ativo", true)
+    .limit(1)
+    .maybeSingle();
+  if (locErr) {
+    console.error("[ghl-backfill] leitura da location falhou:", locErr.message);
+    throw new BackfillErro(500, "Falha ao ler a conta do CRM");
+  }
+  if (!loc?.location_id) throw new BackfillErro(404, "Cliente sem conta do CRM conectada");
+  const locationId = loc.location_id as string;
+
+  const bearer = await credencialDaConta(locationId, fetchImpl);
+
+  const base = (process.env.GHL_API_BASE_URL || "").replace(/\/+$/, "");
+  if (!base) {
+    console.error("[ghl-backfill] Variável de ambiente ausente: GHL_API_BASE_URL");
+    throw new BackfillErro(500, "Integração com o CRM não configurada");
+  }
+
+  let cursor = args.cursor;
+  let processados = 0;
+  let falhas = 0;
+  let terminou = false;
+
+  for (let pagina = 0; pagina < maxPaginas && !terminou; pagina++) {
+    const resp = await fetchImpl(`${base}/contacts/search`, {
+      method: "POST",
+      headers: { Authorization: `Bearer ${bearer}`, Version: GHL_API_VERSION, "content-type": "application/json", accept: "application/json" },
+      body: JSON.stringify({ locationId, pageLimit: CONTATOS_POR_PAGINA, ...(cursor ? { searchAfter: cursor } : {}) }),
+    });
+    if (!resp.ok) {
+      console.error("[ghl-backfill] busca de contatos falhou", resp.status);
+      throw new BackfillErro(502, "Falha ao buscar contatos no CRM", { processados, falhas, cursor });
+    }
+
+    let dados: { contacts?: unknown };
+    try {
+      dados = JSON.parse(await resp.text()) as { contacts?: unknown };
+    } catch {
+      console.error("[ghl-backfill] resposta de contatos ilegivel", resp.status);
+      throw new BackfillErro(502, "Resposta inválida do CRM ao buscar contatos", { processados, falhas, cursor });
+    }
+    const contatos = Array.isArray(dados.contacts) ? (dados.contacts as Record<string, unknown>[]) : [];
+
+    for (const contato of contatos) {
+      try {
+        const { status } = await ingerirLeadGhl(
+          { ...contato, locationId },
+          { mapper: mapGhlAppContactToLead, created_at: dataOriginal(contato.dateAdded) },
+        );
+        if (status === 200 || status === 201) processados++;
+        else {
+          falhas++;
+          console.warn("[ghl-backfill] contato nao gravado", String(contato.id ?? "?"), status);
+        }
+      } catch (err) {
+        falhas++;
+        console.error("[ghl-backfill] contato falhou", String(contato.id ?? "?"), err instanceof Error ? err.message : "erro interno");
+      }
+    }
+
+    const ultimo = contatos[contatos.length - 1];
+    const proximo = ultimo ? lerCursor(ultimo.searchAfter) : null;
+    if (contatos.length < CONTATOS_POR_PAGINA || !proximo) {
+      terminou = true;
+      cursor = null;
+    } else {
+      cursor = proximo;
+    }
+  }
+
+  return { processados, falhas, cursor, terminou };
+}
