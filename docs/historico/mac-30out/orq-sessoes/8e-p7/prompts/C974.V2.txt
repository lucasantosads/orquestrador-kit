+/**
+ * Espaçamento de envios de WhatsApp. Rajada de mensagens de um mesmo número é o
+ * principal gatilho de bloqueio, então cada envio espera um intervalo sorteado
+ * desde o anterior e a execução tem um teto de envios.
+ */
+
+export interface RitmoDeps {
+  agora: () => number;
+  dormir: (ms: number) => Promise<void>;
+  aleatorio: () => number;
+}
+
+export type VezRitmo = { liberado: true } | { liberado: false; motivo: string };
+
+export const MOTIVO_TETO = "teto de envios por execucao atingido";
+
+export function criarRitmo(opts?: {
+  minMs?: number;
+  maxMs?: number;
+  tetoPorProcesso?: number;
+  deps?: Partial<RitmoDeps>;
+}) {
+  const minMs = opts?.minMs ?? 3000;
+  const maxMs = opts?.maxMs ?? 8000;
+  const teto = opts?.tetoPorProcesso ?? 60;
+  const deps: RitmoDeps = {
+    agora: () => Date.now(),
+    dormir: (ms) => new Promise<void>((resolve) => setTimeout(resolve, ms)),
+    aleatorio: () => Math.random(),
+    ...opts?.deps,
+  };
+
+  let enviados = 0;
+  let ultimoEnvio: number | null = null;
+
+  return {
+    async aguardarVez(): Promise<VezRitmo> {
+      if (enviados >= teto) return { liberado: false, motivo: MOTIVO_TETO };
+      enviados += 1;
+      const inicio = deps.agora();
+      let liberadoEm = inicio;
+      if (ultimoEnvio !== null) {
+        const intervalo = minMs + deps.aleatorio() * (maxMs - minMs);
+        const espera = Math.max(0, ultimoEnvio + intervalo - inicio);
+        if (espera > 0) {
+          await deps.dormir(espera);
+          liberadoEm = inicio + espera;
+        }
+      }
+      ultimoEnvio = liberadoEm;
+      return { liberado: true };
+    },
+  };
+}
+
+// Estado em memória do módulo: vale por execução da função serverless.
+// Nos testes (VITEST) não dorme, mas mantém a contagem do teto.
+export const ritmoPadrao = criarRitmo({
+  deps: {
+    dormir: (ms) =>
+      process.env.VITEST ? Promise.resolve() : new Promise<void>((resolve) => setTimeout(resolve, ms)),
+  },
+});
