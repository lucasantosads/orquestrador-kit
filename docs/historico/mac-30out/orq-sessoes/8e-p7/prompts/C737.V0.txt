+              {aviso[chave] ? (
+                <p role="alert" className="text-xs text-destructive">
+                  {aviso[chave]}
+                </p>
+              ) : null}
+            </li>
+          );
+        })}
+      </ul>
+      {desconhecidos > 0 ? (

+          <span className="font-mono">{desconhecidos}</span>{" "}

+          <Link href="/portal/equipe" className="text-primary hover:underline">
+            Identificar
+          </Link>
+        </p>
+      ) : null}
+    </section>
