+const dia = (n: number) => new Date(Date.now() - 3 * 3600000 + n * 86400000).toISOString().slice(0, 10);
+
+beforeEach(() => {
+  banco.falhar.clear();
+  banco.colunas = [];
+  sessao.atual = { employeeId: ID.G1 };
+  banco.tabelas = {
+    ops_cargo: [["cg", "Gestor de tráfego"], ["ch", "Head de operação"], ["cc", "Customer Success"]]
+      .map(([id, nome]) => ({ id, nome, role_base: "operacao", eh_lideranca: id === "ch" })),
+    employees: [[ID.G1, "cg"], [ID.G2, "cg"], [ID.H, "ch"], [ID.CS, "cc"]].map(([id, cargo_id]) => (
+      { id, nome: id, cargo_id, ativo: true, is_head_operacional: cargo_id === "ch" })),
+    clientes: (
+      [[ID.A1, "Alfa Um", "ativo"], [ID.A2, "Alfa Dois", "ativo"], [ID.A3, "Alfa Três", "ativo"], [ID.A4, "Alfa Quatro", "pausado"],
+        [ID.B1, "Beta Um", "ativo"], [ID.X, "Sem Gestor", "ativo"], [ID.Z, "Encerrado", "churned"]] as string[][]
+    ).map(([id, nome, status]) => ({ id, nome, status })),
+    perf_cliente_gestor: [ID.A1, ID.A2, ID.A3, ID.A4, ID.Z]
+      .map((c) => ({ cliente_id: c, gestor_id: ID.G1, ativo: true, principal: true }))
+      .concat([{ cliente_id: ID.B1, gestor_id: ID.G2, ativo: true, principal: true }]),
+    wpp_grupo: [ID.A1, ID.A2, ID.A3, ID.A4, ID.B1, ID.X].map((c, i) => ({ group_jid: `g-${i}`, cliente_id: c, deleted_at: null, status: "ativo" })),
+    cs_wpp_mensagem: [
+      { remote_jid: "g-0", enviada_em: dia(-1) + "T12:00:00.000Z" },
+      { remote_jid: "g-1", enviada_em: dia(0) + "T13:00:00.000Z" }, // última do grupo, de quem quer que seja
+    ],
+    // Renovação: A1 vencida (receita), A2 em 10 dias e A3 em 25 (contrato), A4 sem fidelidade.
+    clientes_receita: [{ cliente_id: ID.A1, fidelidade_fim: dia(-10), mrr: 9000 }],
+    contratos: [
+      { cliente_id: ID.A2, fidelidade_fim: dia(10), data_fechamento: "2026-01-01", valor: 5000 },
+      { cliente_id: ID.A3, fidelidade_fim: dia(25), data_fechamento: "2026-02-01", valor: 5000 },
+    ],
+  };
+});
+
+async function pedir(qs = "") {
+  const res = await GET(new Request("http://localhost/api/portal/inicio" + qs));
+  return { status: res.status, body: (await res.json()) as Record<string, any> }; // eslint-disable-line @typescript-eslint/no-explicit-any
+}
+const nomes = (b: Record<string, any>) => (b.carteira as { nome: string }[]).map((l) => l.nome); // eslint-disable-line @typescript-eslint/no-explicit-any
+
+describe("GET /api/portal/inicio", () => {
+  it("401 sem sessão", async () => {
+    sessao.atual = null;
+    expect((await pedir()).status).toBe(401);
+  });
+
+  it("dois gestores recebem só os clientes de cada um", async () => {
+    const g1 = await pedir();
+    expect(g1.status).toBe(200);
+    expect(nomes(g1.body)).toContain("Alfa Um");
+    expect(nomes(g1.body)).toContain("Alfa Quatro");
+    expect(nomes(g1.body)).not.toContain("Beta Um");
+    expect(nomes(g1.body)).not.toContain("Sem Gestor");
+    expect(nomes(g1.body)).not.toContain("Encerrado");
+    expect(g1.body.carteira).toHaveLength(4);
+
+    sessao.atual = { employeeId: ID.G2 };
+    const g2 = await pedir();
+    expect(nomes(g2.body)).toContain("Beta Um");
+    expect(nomes(g2.body)).not.toContain("Alfa Um");
+    expect(g2.body.carteira).toHaveLength(1);
+    // vácuo contado só na carteira de quem pede
+    expect(g2.body.contadores.grupos_esperando).toEqual({ valor: 1, mais_antigo_min: 70 });
+  });
+
+  it("head vê os clientes sem gestor; CS vê todos os ativos", async () => {
+    sessao.atual = { employeeId: ID.H };
+    const h = await pedir();
+    expect(nomes(h.body)).toContain("Sem Gestor");
+    expect(nomes(h.body)).not.toContain("Alfa Um");
+
+    sessao.atual = { employeeId: ID.CS };
+    const cs = await pedir();
+    expect(cs.body.carteira).toHaveLength(6);
+    expect(nomes(cs.body)).not.toContain("Encerrado");
+  });
+
+  it("contador de risco = vermelhos + amarelos da própria carteira", async () => {
+    const { body } = await pedir();
+    const cores = (body.carteira as { cor: string }[]).map((l) => l.cor);
+    const v = cores.filter((c) => c === "vermelho").length;
+    const a = cores.filter((c) => c === "amarelo").length;
+    // A1 vácuo de 500 min = vermelho; A4 vácuo de 90 min = amarelo; A2 e A3 sem sinal.
+    expect(body.contadores.risco).toEqual({ valor: v + a, vermelhos: 1, amarelos: 1, de: 4 });
+    expect(body.contadores.grupos_esperando).toEqual({ valor: 2, mais_antigo_min: 500 });
+  });
+
+  it("renovações contam vencida, 15, 30 e sem fidelidade", async () => {
+    const { body } = await pedir();
+    expect(body.contadores.renovacoes).toEqual({ em_30: 1, em_15: 1, vencidas: 1, sem_fidelidade: 1 });
+  });
+
+  it("carimbo é a última mensagem do grupo; promessa sai sem base, nunca zero", async () => {
+    const { body } = await pedir();
+    expect(body.atualizado.ultima_mensagem_grupo_em).toBe(dia(0) + "T13:00:00.000Z");
+    expect(body.contadores.promessas.estado).toBe("sem_base");
+    expect(body.contadores.promessas.motivo).toContain("promessas");
+  });
+
+  it("falha de leitura = 500 com o motivo, nunca carteira vazia", async () => {
+    banco.falhar.add("perf_ad_accounts");
+    const { status, body } = await pedir();
+    expect(status).toBe(500);
+    expect(body.error).toContain("contas de anúncio");
+    expect(body.carteira).toBeUndefined();
+  });
+
+  it("resposta não tem as chaves mrr nem valor de contrato", async () => {
+    const { body } = await pedir();
+    expect(Object.keys(body)).not.toContain("mrr");
+    expect(Object.keys(body)).not.toContain("valor");
+    for (const l of body.carteira as Linha[]) {
+      expect(Object.keys(l)).not.toContain("mrr");
+      expect(Object.keys(l)).not.toContain("valor");
+    }
+    expect(banco.colunas).not.toContain("mrr");
+    expect(banco.colunas).not.toContain("valor_contrato");
+  });
+
+  it("view_as de gestor é ignorado; de head é honrado", async () => {
+    const g1 = await pedir(`?view_as=${ID.G2}`);
+    expect(nomes(g1.body)).toContain("Alfa Um");
+    expect(nomes(g1.body)).not.toContain("Beta Um");
+
+    sessao.atual = { employeeId: ID.H };
+    const h = await pedir(`?view_as=${ID.G2}`);
+    expect(nomes(h.body)).toContain("Beta Um");
+    expect(h.body.carteira).toHaveLength(1);
+  });
+});
