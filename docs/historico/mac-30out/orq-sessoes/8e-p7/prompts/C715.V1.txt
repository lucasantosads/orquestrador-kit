+/**
+ * Job periódico do portal do gestor (ticket 715): roda as etapas que alimentam
+ * os eventos de produtividade. Cada etapa tem try/catch próprio, então uma que
+ * falha não impede as outras. Chamado por src/app/api/cron/operacao-inicio.
+ */
+import type { Db } from "@/lib/clientes/relacao";
+import { derivarRespostas, JANELA_HORAS } from "@/lib/eventos/resposta-grupo";
+
+export interface EtapaOperacao {
+  nome: string;
+  rodar: (db: Db, agoraISO: string) => Promise<string>;
+}
+
+export interface ResultadoEtapa {
+  nome: string;
+  ok: boolean;
+  resumo: string | null;
+  erro: string | null;
+}
+
+export const ETAPAS: EtapaOperacao[] = [
+  {
+    nome: "eventos_resposta",
+    rodar: async (db, agoraISO) => {
+      const desde = new Date(Date.parse(agoraISO) - JANELA_HORAS * 3_600_000).toISOString();
+      const r = await derivarRespostas(db, desde, agoraISO);
+      return `${r.pares} pares, ${r.gravados} gravados, ${r.falhas} falhas`;
+    },
+  },
+];
+
+export async function rodarOperacaoInicio(
+  db: Db,
+  agoraISO: string,
+  etapas: EtapaOperacao[] = ETAPAS,
+): Promise<{ etapas: ResultadoEtapa[] }> {
+  const resultados: ResultadoEtapa[] = [];
+  for (const e of etapas) {
+    try {
+      resultados.push({ nome: e.nome, ok: true, resumo: await e.rodar(db, agoraISO), erro: null });
+    } catch (err) {
+      resultados.push({ nome: e.nome, ok: false, resumo: null, erro: err instanceof Error ? err.message : String(err) });
+    }
+  }
+  return { etapas: resultados };

+        return Promise.resolve({ error: null });
