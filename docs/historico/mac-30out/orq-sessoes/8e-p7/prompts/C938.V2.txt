+/**
+ * Verba recomendada pela tese (ticket 938). Regra pura: sem banco, sem fetch.
+ * A verba é EXIBIDA, nunca gravada: só o clique em salvar do gestor grava.
+ * Dinheiro aqui é do CLIENTE (verba de anúncio), não da agência.
+ */
+import type { LinhaTese } from "./agregar";
+
+export interface MetaMensalEntrada {
+  metrica_ancora: "faturamento" | "contratos";
+  meta_valor: number;
+  ticket_medio_alvo: number | null;
+}
+
+export interface EntradaVerba {
+  meta: MetaMensalEntrada | null;
+  tese: LinhaTese | null;
+  /** Ticket realizado da tese (faturamento gerado / contratos); usado só sem ticket na meta. */
+  ticket_tese: number | null;
+}
+
+export interface VerbaRecomendada {
+  valor: number | null;
+  contratos: number | null;
+  cac_tese: number | null;
+  conta: string;
+  motivo: string | null;
+}
+
+// \s casa o espaço não separável que o Intl põe depois do R$.
+const brl = (n: number): string => n.toLocaleString("pt-BR", { style: "currency", currency: "BRL" }).replace(/\s/g, " ");
+
+export function recomendarVerba({ meta, tese, ticket_tese }: EntradaVerba): VerbaRecomendada {
+  const vazio = { valor: null, contratos: null, cac_tese: null, conta: "" } as const;
+  if (!meta || !(meta.meta_valor > 0)) return { ...vazio, motivo: "cliente sem meta de contratos ou faturamento no mes" };
+  if (!tese) return { ...vazio, motivo: "cliente sem campanha com tese" };
+  if (tese.sem_amostra) return { ...vazio, motivo: `sem amostra na tese: ${tese.clientes} clientes` };
+
+  const cac = tese.cac.valor;
+  if (cac === null || !(cac > 0)) return { ...vazio, motivo: "sem base: a tese nao tem venda registrada no periodo" };
+
+  let contratos: number;
+  if (meta.metrica_ancora === "contratos") {
+    contratos = meta.meta_valor;
+  } else {
+    const ticket = meta.ticket_medio_alvo ?? ticket_tese;
+    if (ticket === null || !(ticket > 0)) return { ...vazio, motivo: "sem ticket para converter faturamento em contratos" };
+    contratos = Math.ceil(meta.meta_valor / ticket);
+  }
+
+  const valor = Math.round(contratos * cac * 100) / 100;
+  const conta = `${brl(valor)} = ${contratos} contratos x CAC mediano ${brl(cac)} de ${tese.nome} (${tese.clientes} clientes)`;
+  return { valor, contratos, cac_tese: cac, conta, motivo: null };
+}
