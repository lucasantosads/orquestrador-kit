+function ler<T>(oque: string, r: { data: unknown; error: { message?: string } | null }): T[] {
+  if (r.error) throw new Error(`${oque}: ${r.error.message ?? "erro de leitura"}`);
+  return (r.data ?? []) as T[];

+export async function GET() {
+  const session = await getSession();
+  if (!session) return NextResponse.json({ error: "Não autenticado" }, { status: 401 });
+  try {
+    const [empRes, cargoRes] = await Promise.all([

+      supabase.from("ops_cargo").select("id, nome, role_base, eh_lideranca"),
+    ]);

+    }));

+    if (!quem || VEEM_EQUIPE.indexOf(quem.papel) === -1) {
+      return NextResponse.json({ error: "Sem permissão para ver a equipe" }, { status: 403 });
+    }
+
+    const agora = new Date().toISOString();
+    const hoje = new Date(Date.now() - 3 * 3600000).toISOString().slice(0, 10);
+    const gestores = equipe.filter((x) => x.papel === "gestor" && x.e.ativo !== false);
+    const gestorIds = gestores.map((g) => g.e.id);
+    const [q, clientesRes, vacuosRes, tarefasRes] = await Promise.all([
+      carregarQuemCuida(supabase),
+      supabase.from("clientes").select("id, nome, status"),
+      carregarVacuos(supabase, agora),
+      supabase.from("tarefas_kanban").select("responsavel_id, status, data_vencimento")
+        .in("responsavel_id", gestorIds).is("deleted_at", null).lte("data_vencimento", hoje),
+    ]);
+    const clientes = ler<{ id: string; nome: string | null; status: string | null }>("clientes", clientesRes)
+      .filter((c) => isClienteAtivo(c.status));
+    const nomeDe = new Map(clientes.map((c) => [c.id, c.nome ?? ""] as const));
+    const atrasadas = ler<{ responsavel_id: string; status: string | null; data_vencimento: string | null }>("tarefas", tarefasRes)
+      .filter((t) => t.status !== "concluido" && (t.data_vencimento ?? "9") < hoje);
+    const vacuos = vacuosRes.vacuos.filter((v) => nomeDe.has(v.cliente_id));
+    const contar = (ids: string[], min: number) =>
+      vacuos.filter((v) => ids.indexOf(v.cliente_id) !== -1 && v.minutos_uteis > min).length;
+
+    const resumos: ResumoInicio[] = await Promise.all(
+      gestores.map((g) => carregarInicio(supabase, { id: g.e.id, papel: "gestor" }, agora)),
+    );
+    const linhas: Linha[] = gestores.map((g, i) => {
+      const r = resumos[i];
+      const ids = r.carteira.map((l) => l.cliente_id);
+      const respostaMedia: number | SemBase = r.resposta_media_min ?? { estado: "sem_base", motivo: "menos de 5 respostas registradas" };
+      return {
+        gestor_id: g.e.id, nome: g.e.nome ?? "", clientes: r.carteira.length,
+        esperando: r.contadores.grupos_esperando.valor, mais_de_1_dia: contar(ids, LIMITE_FILA_MIN),
+        promessas: SEM_PROMESSAS, vermelhos: r.contadores.risco.vermelhos,
+        acoes_atrasadas: atrasadas.filter((t) => t.responsavel_id === g.e.id).length, resposta_media_min: respostaMedia,
+      };
+    });
+    linhas.sort((a, b) => (b.esperando as number) - (a.esperando as number));
+
+    // Sem gestor: clientes que hoje caem no head. A cor vem do Início de um head (mesma função).
+    const semGestorIds = clientes.filter((c) => responsaveisDoCliente(q, c.id).origem === "head").map((c) => c.id);
+    let vermelhosSem: number | SemBase = { estado: "sem_base", motivo: "nenhum head ativo para calcular" };
+    if (q.heads.length > 0) {
+      const rh = await carregarInicio(supabase, { id: q.heads[0].id, papel: "head" }, agora);
+      vermelhosSem = rh.carteira.filter((l) => semGestorIds.indexOf(l.cliente_id) !== -1 && l.cor === "vermelho").length;
+    }
+    const naoSeAplica: SemBase = { estado: "sem_base", motivo: "sem gestor responsável" };
+    const sem_gestor = {
+      clientes: semGestorIds.length, esperando: contar(semGestorIds, -1), mais_de_1_dia: contar(semGestorIds, LIMITE_FILA_MIN),
+      promessas: SEM_PROMESSAS, vermelhos: vermelhosSem, acoes_atrasadas: naoSeAplica, resposta_media_min: naoSeAplica,
+    };
+
+    const fila = vacuos.filter((v) => v.minutos_uteis > LIMITE_FILA_MIN)
+      .sort((a, b) => b.minutos_uteis - a.minutos_uteis)
+      .map((v) => {
+        const r = responsaveisDoCliente(q, v.cliente_id);
+        const g = r.origem === "gestor" ? r.pessoas[0] : null;
+        return {
+          tipo: "vacuo" as const, cliente_id: v.cliente_id, cliente_nome: nomeDe.get(v.cliente_id) ?? "", group_jid: v.group_jid,
+          desde: v.desde, minutos_uteis: v.minutos_uteis, gestor_id: g?.id ?? null, gestor_nome: g?.nome ?? "Sem gestor",
+        };
+      });
+    return NextResponse.json({ gestores: linhas, sem_gestor, fila, promessas_vencidas: SEM_PROMESSAS });
+  } catch (e) {
+    const motivo = e instanceof Error ? e.message : String(e);
+    return NextResponse.json({ error: `Não foi possível montar a equipe: ${motivo}` }, { status: 500 });
+  }

+        neq: (k: string, v: unknown) => f((r) => r[k] !== v),
+        is: (k: string, v: unknown) => f((r) => (r[k] ?? null) === v),
+        in: (k: string, vs: unknown[]) => f((r) => vs.includes(r[k])),
+        not: (k: string) => f((r) => r[k] != null),
+        gte: (k: string, v: string) => f((r) => String(r[k]) >= v),
+        lte: (k: string, v: string) => f((r) => String(r[k]) <= v),
+        order: () => q,

+        maybeSingle: async () => ({ data: exec().data[0] ?? null, error: null }),
+        then: (ok: (v: unknown) => unknown, ko?: (e: unknown) => unknown) => Promise.resolve(exec()).then(ok, ko),
